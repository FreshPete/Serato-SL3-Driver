cmake_minimum_required(VERSION 3.15)
project(sl3_descriptor_pair LANGUAGES CXX)

if(NOT MSVC OR NOT CMAKE_VS_PLATFORM_NAME STREQUAL "x64")
  message(FATAL_ERROR "The descriptor pair helper requires MSVC and an x64 build")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(sl3_descriptor_pair SHARED descriptor-pair.cpp)
add_executable(sl3_descriptor_pair_tests descriptor-pair-tests.cpp)
target_link_libraries(sl3_descriptor_pair_tests PRIVATE sl3_descriptor_pair)
add_library(sl3_descriptor_reader SHARED descriptor-reader.cpp)
target_compile_definitions(sl3_descriptor_reader PRIVATE SL3_DESCRIPTOR_READER_EXPORT)
add_executable(sl3_descriptor_reader_tests descriptor-reader-tests.cpp)
target_link_libraries(sl3_descriptor_reader_tests PRIVATE sl3_descriptor_reader)
add_library(sl3_descriptor_factory SHARED descriptor-factory.cpp)
target_compile_definitions(sl3_descriptor_factory PRIVATE SL3_DESCRIPTOR_FACTORY_EXPORT)
add_executable(sl3_descriptor_factory_tests descriptor-factory-tests.cpp)
target_link_libraries(sl3_descriptor_factory_tests PRIVATE sl3_descriptor_factory)
foreach(target sl3_descriptor_pair sl3_descriptor_pair_tests sl3_descriptor_reader sl3_descriptor_reader_tests
               sl3_descriptor_factory sl3_descriptor_factory_tests)
  target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(${target} PRIVATE /W4 /WX /permissive- /guard:cf)
  target_link_options(${target} PRIVATE /guard:cf)
endforeach()

enable_testing()
add_test(NAME sl3_descriptor_pair_tests COMMAND sl3_descriptor_pair_tests)
add_test(NAME sl3_descriptor_reader_tests COMMAND sl3_descriptor_reader_tests)
add_test(NAME sl3_descriptor_factory_tests COMMAND sl3_descriptor_factory_tests)

set(SL3_GUM_SOURCE "" CACHE PATH "Existing Frida Gum source directory for the native hook fixture")
set(SL3_GUM_ARCHIVE "" CACHE FILEPATH "Existing Windows x64 Frida Gum static library")
set(SL3_GUM_SDK "" CACHE PATH "Windows x64 static Frida SDK used to build Gum")
if(SL3_GUM_SOURCE AND SL3_GUM_ARCHIVE AND SL3_GUM_SDK)
  get_filename_component(gum_build_dir "${SL3_GUM_ARCHIVE}" DIRECTORY)
  get_filename_component(gum_build_dir "${gum_build_dir}" DIRECTORY)
  add_library(sl3_descriptor_native_hooks STATIC descriptor-native-hooks.cpp)
  target_include_directories(sl3_descriptor_native_hooks SYSTEM PUBLIC
    "${SL3_GUM_SOURCE}" "${gum_build_dir}" "${SL3_GUM_SDK}/include" "${SL3_GUM_SDK}/include/glib-2.0"
    "${SL3_GUM_SDK}/lib/glib-2.0/include" "${SL3_GUM_SDK}/include/capstone")
  target_compile_definitions(sl3_descriptor_native_hooks PUBLIC GUM_STATIC)
  target_link_libraries(sl3_descriptor_native_hooks PUBLIC "${SL3_GUM_ARCHIVE}")
  foreach(component gio-2.0 gobject-2.0 gmodule-2.0 glib-2.0 pcre2-8 ffi z capstone json-glib-1.0 tcc lzma)
    target_link_libraries(sl3_descriptor_native_hooks PUBLIC "${SL3_GUM_SDK}/lib/lib${component}.a")
  endforeach()
  target_link_libraries(sl3_descriptor_native_hooks PUBLIC
    advapi32 crypt32 dbghelp dnsapi iphlpapi ole32 psapi shlwapi user32 winmm ws2_32)
  add_executable(sl3_descriptor_native_hooks_tests descriptor-native-hooks-tests.cpp)
  target_link_libraries(sl3_descriptor_native_hooks_tests PRIVATE sl3_descriptor_native_hooks sl3_descriptor_reader)
  add_library(sl3_descriptor_hook_library SHARED descriptor-hook-library.cpp
    descriptor-factory.cpp descriptor-reader.cpp descriptor-pair.cpp)
  target_compile_definitions(sl3_descriptor_hook_library PRIVATE SL3_DESCRIPTOR_FACTORY_EXPORT SL3_DESCRIPTOR_READER_EXPORT)
  target_link_libraries(sl3_descriptor_hook_library PRIVATE sl3_descriptor_native_hooks)
  foreach(target sl3_descriptor_native_hooks sl3_descriptor_native_hooks_tests sl3_descriptor_hook_library)
    set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
    target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /guard:cf)
    target_link_options(${target} PRIVATE /guard:cf)
  endforeach()
  add_test(NAME sl3_descriptor_native_hooks_tests COMMAND sl3_descriptor_native_hooks_tests)
endif()
