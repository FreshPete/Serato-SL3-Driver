cmake_minimum_required(VERSION 3.20)
project(sl3_setup_window LANGUAGES CXX RC)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(SL3_Compatibility_Setup WIN32 "${CMAKE_CURRENT_SOURCE_DIR}/../../../src/installer/SetupWindow.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/../../../src/installer/SetupWindow.rc")
add_executable(sl3_setup_window_tests "${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/native/setup_window_tests.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/../../../src/installer/SetupWindow.rc")
foreach(target SL3_Compatibility_Setup sl3_setup_window_tests)
 target_compile_features(${target} PRIVATE cxx_std_17)
 target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
 target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4;/WX;/permissive-;/EHsc;/utf-8>)
 target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../../../src/installer")
 target_link_libraries(${target} PRIVATE windowsapp shell32 ole32 bcrypt advapi32 user32 gdi32 comctl32)
endforeach()
enable_testing()
add_test(NAME setup_window_contract COMMAND sl3_setup_window_tests)
set_tests_properties(setup_window_contract PROPERTIES TIMEOUT 120)
