cmake_minimum_required(VERSION 3.20)
project(sl3_native_silent_duplex LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 diagnostic only")
endif()
get_filename_component(SL3_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SDK "${SL3_ROOT}/artifacts/references/asio-sdk-2.3.4/ASIOSDK/common")
file(SHA256 "${SDK}/iasiodrv.h" ABI_SHA)
if(NOT ABI_SHA STREQUAL "86ad4788452c74823653cf57d4bd557c48bb6542d7e6eb7fff80102c705a066d")
  message(FATAL_ERROR "Reviewed ASIO interface required")
endif()
add_executable(sl3_native_silent_duplex_host host.cpp)
target_compile_features(sl3_native_silent_duplex_host PRIVATE cxx_std_17)
target_include_directories(sl3_native_silent_duplex_host PRIVATE "${SL3_ROOT}/src" "${SDK}")
target_compile_definitions(sl3_native_silent_duplex_host PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(sl3_native_silent_duplex_host PRIVATE /W4 /WX /permissive- /EHsc)
target_link_libraries(sl3_native_silent_duplex_host PRIVATE ole32 uuid)
enable_testing()
add_test(NAME native_silent_plan COMMAND sl3_native_silent_duplex_host --plan)
add_test(NAME native_silent_unknown COMMAND sl3_native_silent_duplex_host --unknown)
add_test(NAME native_reopen_requires_private_paths COMMAND sl3_native_silent_duplex_host --run-native-application-reopen)
set_tests_properties(native_silent_unknown native_reopen_requires_private_paths PROPERTIES WILL_FAIL TRUE)
set_tests_properties(native_silent_plan native_silent_unknown native_reopen_requires_private_paths PROPERTIES TIMEOUT 10)
