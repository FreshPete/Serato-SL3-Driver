cmake_minimum_required(VERSION 3.20)
project(sl3_asio_private_trace LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 diagnostic only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(SL3_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SDK "${SL3_ROOT}/artifacts/references/asio-sdk-2.3.4/ASIOSDK/common")
file(SHA256 "${SDK}/iasiodrv.h" ABI_SHA)
if(NOT ABI_SHA STREQUAL "86ad4788452c74823653cf57d4bd557c48bb6542d7e6eb7fff80102c705a066d")
  message(FATAL_ERROR "Reviewed official ASIO interface required")
endif()
add_executable(sl3_asio_private_trace private_capture.cpp)
target_include_directories(sl3_asio_private_trace PRIVATE "${SL3_ROOT}/src" "${SDK}")
target_compile_definitions(sl3_asio_private_trace PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(sl3_asio_private_trace PRIVATE /W4 /WX /permissive- /EHsc)
target_link_libraries(sl3_asio_private_trace PRIVATE advapi32 ole32 uuid)
enable_testing()
add_test(NAME private_trace_plan COMMAND sl3_asio_private_trace --plan)
add_test(NAME private_trace_rejects_unknown COMMAND sl3_asio_private_trace --unknown)
set_tests_properties(private_trace_plan private_trace_rejects_unknown PROPERTIES TIMEOUT 10)
set_tests_properties(private_trace_rejects_unknown PROPERTIES WILL_FAIL TRUE)
