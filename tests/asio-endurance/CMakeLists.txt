cmake_minimum_required(VERSION 3.20)
project(sl3_asio_restart_endurance LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT DEFINED SL3_EXISTING_BUILD OR NOT DEFINED SL3_ASIO_COMMON)
  message(FATAL_ERROR "Windows MSVC, existing Release native build and local ASIO common headers required")
endif()
get_filename_component(SL3_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(sl3_asio_restart_endurance ../native/asio_restart_endurance.cpp)
target_compile_features(sl3_asio_restart_endurance PRIVATE cxx_std_17)
target_compile_definitions(sl3_asio_restart_endurance PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(sl3_asio_restart_endurance PRIVATE /W4 /WX /permissive- /EHsc)
target_include_directories(sl3_asio_restart_endurance PRIVATE "${SL3_ROOT}/src" "${SL3_ASIO_COMMON}")
target_link_libraries(sl3_asio_restart_endurance PRIVATE "${SL3_EXISTING_BUILD}/Release/sl3_worker_process.lib" ole32 uuid bcrypt)
