cmake_minimum_required(VERSION 3.20)
project(sl3_asio_etl_decoder LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 decoder only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(SL3_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(sl3_asio_decode_trace decode.cpp)
target_include_directories(sl3_asio_decode_trace PRIVATE "${SL3_ROOT}/src")
target_compile_definitions(sl3_asio_decode_trace PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(sl3_asio_decode_trace PRIVATE /W4 /WX /permissive- /EHsc)
target_link_libraries(sl3_asio_decode_trace PRIVATE advapi32)
enable_testing()
add_test(NAME decode_plan COMMAND sl3_asio_decode_trace --plan)
add_test(NAME decode_rejects_unknown COMMAND sl3_asio_decode_trace --unknown)
set_tests_properties(decode_plan decode_rejects_unknown PROPERTIES TIMEOUT 10)
set_tests_properties(decode_rejects_unknown PROPERTIES WILL_FAIL TRUE)
