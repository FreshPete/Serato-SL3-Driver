cmake_minimum_required(VERSION 3.20)
project(sl3_setup_entry LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
message(FATAL_ERROR "Windows x64 MSVC required")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(SL3_Setup_Engineering "${CMAKE_CURRENT_SOURCE_DIR}/../../../src/installer/OnDemandKitEntry.cpp")
target_compile_features(SL3_Setup_Engineering PRIVATE cxx_std_17)
target_compile_definitions(SL3_Setup_Engineering PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(SL3_Setup_Engineering PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
target_include_directories(SL3_Setup_Engineering PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(SL3_Setup_Engineering PRIVATE windowsapp shell32 bcrypt advapi32)
add_executable(sl3_setup_result_pipe_tests "${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/native/setup_result_pipe_tests.cpp")
target_compile_features(sl3_setup_result_pipe_tests PRIVATE cxx_std_17)
target_compile_definitions(sl3_setup_result_pipe_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(sl3_setup_result_pipe_tests PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
target_include_directories(sl3_setup_result_pipe_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(sl3_setup_result_pipe_tests PRIVATE windowsapp shell32 bcrypt advapi32)
enable_testing()
add_test(NAME setup_help COMMAND SL3_Setup_Engineering --help)
add_test(NAME setup_unknown COMMAND SL3_Setup_Engineering --install)
set_tests_properties(setup_unknown PROPERTIES WILL_FAIL TRUE)
add_test(NAME setup_result_pipe_contract COMMAND sl3_setup_result_pipe_tests)
set_tests_properties(setup_result_pipe_contract PROPERTIES TIMEOUT 30)
