cmake_minimum_required(VERSION 3.20)
project(sl3_com_registration LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 owned diagnostic only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(SL3_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SDK "${SL3_ROOT}/artifacts/references/asio-sdk-2.3.4/ASIOSDK/common")
file(SHA256 "${SDK}/iasiodrv.h" ABI_SHA)
if(NOT ABI_SHA STREQUAL "86ad4788452c74823653cf57d4bd557c48bb6542d7e6eb7fff80102c705a066d")
  message(FATAL_ERROR "Reviewed official ASIO interface required")
endif()
add_executable(sl3_com_registration_host host.cpp)
target_include_directories(sl3_com_registration_host PRIVATE "${SL3_ROOT}/src" "${SDK}")
target_compile_definitions(sl3_com_registration_host PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(sl3_com_registration_host PRIVATE /W4 /WX /permissive- /EHsc)
target_link_libraries(sl3_com_registration_host PRIVATE ole32 uuid advapi32)
enable_testing()
add_test(NAME com_host_plan COMMAND sl3_com_registration_host --plan)
add_test(NAME com_host_refuses_unknown COMMAND sl3_com_registration_host --unknown)
set_tests_properties(com_host_plan com_host_refuses_unknown PROPERTIES TIMEOUT 10)
set_tests_properties(com_host_refuses_unknown PROPERTIES WILL_FAIL TRUE)
